const express = require('express'); const path = require('path'); const app = express(); const PORT = process.env.PORT || 3000; app.use(express.json()); app.use(express.urlencoded({ extended: true })); // عداد إجمالي مرات التشغيل/الطلبات let totalRequests = 0; // مصفوفات التخزين للمفاتيح والمحظورين let keys = [ { key: "FRX-1QR9SA-HTQ7FE", duration: "Lifetime", scriptName: "Main Script", active: true } ]; let bannedUsers = []; let recentLogs = [ { text: "تم تشغيل السيرفر بنجاح", time: "الآن" } ]; // قراءة ملف index.html وإظهاره في الصفحة الرئيسية (لوحة التحكم) app.get('/', (req, res) => { totalRequests++; res.sendFile(path.join(__dirname, 'index.html')); }); // API لإرسال الإحصائيات الحقيقية للوحة التحكم app.get('/api/stats', (req, res) => { res.json({ totalRequests: totalRequests, activeKeysCount: keys.filter(k => k.active).length, bannedCount: bannedUsers.length, logs: recentLogs }); }); // API التحقق من المفتاح - يُرجع كود Lua ينفذه روبلوكس app.get('/api/verify', (req, res) => { totalRequests++; const key = req.query.key; const found = keys.find(k => k.key === key && k.active); // ضبط صيغة الاستجابة إلى نص صافي يفهمه loadstring res.setHeader('Content-Type', 'text/plain'); if (found) { recentLogs.unshift({ text: `تم التحقق من مفتاح بنجاح: ${key}`, time: "منذ قليل" }); res.send(` game.StarterGui:SetCore("SendNotification", { Title = "FRX LUA"; Text = "تم التحقق من المفتاح بنجاح! أهلاً بك."; Duration = 5; }); print("FRX LUA Loaded Successfully!"); `); } else { recentLogs.unshift({ text: `محاولة دخول بمفتاح خاطئ أو موقوف`, time: "منذ قليل" }); res.status(403).send(` game.StarterGui:SetCore("SendNotification", { Title = "FRX LUA"; Text = "المفتاح غير صالح أو غير مفعل!"; Duration = 5; }); error("Invalid or expired key!"); `); } }); // API إضافة مفتاح جديد من لوحة التحكم app.post('/api/add-key', (req, res) => { const { key, duration, scriptName } = req.body; keys.push({ key, duration, scriptName, active: true }); recentLogs.unshift({ text: `تم إضافة مفتاح جديد (${scriptName || 'سكربت'})`, time: "منذ قليل" }); res.json({ success: true, message: "Key added successfully!" }); }); // API حظر مستخدم app.post('/api/ban', (req, res) => { const { userId } = req.body; bannedUsers.push(userId); recentLogs.unshift({ text: `تم حظر المستخدم ID: ${userId}`, time: "منذ قليل" }); res.json({ success: true, message: "User banned successfully!" }); }); app.listen(PORT, () => { console.log(`Server is running on port ${PORT}`); });
